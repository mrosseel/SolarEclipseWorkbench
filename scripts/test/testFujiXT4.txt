# Bench test for the Fujifilm X-T4 over USB only - no relay.
#
# Run it from the GUI with a simulated C1 about two minutes ahead.  Everything is
# scheduled relative to C1 and the test finishes 10 s after it.
#
# Camera state before you start (see fuji-xt4.md for the full list):
#   mode / AE          Manual.  Aperture Priority is what made the first run fail:
#                      the body owns the shutter there and every exposure below is
#                      ignored.  validate_for_eclipse() reports this at detection.
#   focus selector     M
#   shutter speed dial T, so the SDK owns shutter speed
#   ISO dial           C, so the SDK owns ISO
#   aperture           on an XF lens, put the aperture ring at A or the SDK cannot
#                      set f-stops.  On the telescope there is no aperture at all
#                      and the column below is informational.
#
# If the previous run ended in 0x1006 "Camera is busy" or "camera may need power
# cycle", power cycle the body and replug USB before running this.  A wedged SDK
# pipeline does not recover on its own.
#
# What it checks, in order:
#   1. the body answers and the clock syncs
#   2. shutter speed and ISO changes actually reach the camera between frames
#   3. take_burst - for Fuji the argument is a FRAME COUNT, not seconds
#   4. take_bracket - the SDK bracket path the production script relies on
#   5. multi-second exposures work from dial position T, so bulb is not needed
#
# take_hdr is deliberately absent.  On the Fuji SDK path it falls through to the
# virtual-camera branch and fires 2*stops+1 frames at ONE unchanged speed - the
# 14:29 log shows exactly that, "Virtual camera HDR capture failed".  Use
# take_bracket on this body; take_hdr is for the gphoto2 cameras.

voice_prompt, C1, -, 0:01:40.0, C1_IN_60_SECONDS, "X-T4 USB test starting"
sync_cameras, C1, -, 0:01:30.0, "Sync the camera clock and settings"

# --- Single frames: does a settings change actually reach the body? ---
take_picture, C1, -, 0:01:20.0, Fuji Fujifilm X-T4, 1/1000, 8, 400, "Single shot, baseline"
take_picture, C1, -, 0:01:12.0, Fuji Fujifilm X-T4, 1/4000, 8, 800, "Single shot, shutter and ISO both changed"
take_picture, C1, -, 0:01:04.0, Fuji Fujifilm X-T4, 1/250, 8, 200, "Single shot, changed back the other way"

# --- Burst: the last argument is a frame count on Fuji, not a duration ---
take_burst, C1, -, 0:00:55.0, Fuji Fujifilm X-T4, 1/2000, 8, 400, 5, "Burst of 5 frames, expect about 3 s at the 1.8 fps SDK ceiling"

# --- Bracket: what the production script uses for the corona ladders ---
take_bracket, C1, -, 0:00:45.0, Fuji Fujifilm X-T4, 1/500, 8, 400, +/- 1, "Bracket +/- 1 EV, 7 frames at 1/3 stops"
take_bracket, C1, -, 0:00:30.0, Fuji Fujifilm X-T4, 1/250, 8, 400, +/- 2, "Bracket +/- 2 EV, 13 frames - the production corona bracket"

# --- Long exposures from dial position T, no bulb required ---
take_picture, C1, -, 0:00:14.0, Fuji Fujifilm X-T4, 1, 8, 800, "One second, as used for the deep corona"
take_picture, C1, -, 0:00:09.0, Fuji Fujifilm X-T4, 1.6, 8, 800, "1.6 seconds, the longest frame in the production script"

# --- Contact moment ---
voice_prompt, C1, -, 0:00:05.0, C1_IN_5_SECONDS, "Five seconds"
voice_prompt, C1, -, 0:00:00.0, C1, "First contact"
take_picture, C1, +, 0:00:03.0, Fuji Fujifilm X-T4, 1/1000, 8, 400, "Single shot on the contact moment"
take_picture, C1, +, 0:00:10.0, Fuji Fujifilm X-T4, 1/1000, 8, 400, "Final frame - check the count and the EXIF"
